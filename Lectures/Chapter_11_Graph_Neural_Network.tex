\begin{frame}{What is Graph?}

\footnotesize

\begin{center}
A graph $G$ is a data structure described by a set of edges and vertices (also known as nodes):

$$
G = (V, E) \text{~~~ or ~~~~} G = (V, E, A)
$$
\end{center}


\begin{itemize}
\item $V$ set of vertices.
\item $E$ set of edges.
\item They can be directed or undirected.
\end{itemize}

\begin{center}
A graph is often represented by an Adjacency matrix, $A$. If a graph has $N$ nodes, then $A$ has a dimension of $(N\times N)$. People sometimes provide another feature matrix to describe the nodes in the graph. If each node has F numbers of features, then the feature matrix $X$ has a dimension of $(N\times F)$.

\begin{frame}{Challenges in handling Non-Euclidean Data}

\footnotesize

\begin{itemize}
\item fixed Grid Assumptions in MLPs, CNNs, RNNs
\begin{itemize}
\item Assume regular, structured data (e.g., grids or
sequences).
\item Cannot directly handle irregular neighborhoods or
variable node connectivity in graphs or other
non-Euclidean structures.
\end{itemize}
\item Irregular Neighborhoods:
\begin{itemize}
\item Varying numbers of neighbors per node.
\item No uniform notion of proximity or direction.
\item Standard convolution filters (which operate on fixed local
neighborhoods) fail to adapt to these variable structures.
\end{itemize}
\item Lack of Spatial Regularity:

\begin{itemize}
\item The concept of "locality" is not fixed and varies across the
structure.
\item Non-Euclidean data like point clouds, graphs (undirected)
is unordered.
\end{itemize}
\end{itemize}

\begin{center}
We need a permutation invariance / equivariance.
\end{center}

\end{frame}

\begin{frame}{Curse of Dimensionality}


\footnotesize

\begin{columns}

\column{0.8\linewidth}


In high dimensions, Euclidean distance between data becomes meaningless.

\vspace{10pt}

If data are uniformly distributed in $\Rbb^d$,then pick any data point $\xbf_i \in V$, that gives

\begin{align*}
\lim_{d\to \infty} \Ebb_{x_i}  \bigg( \frac{d_{max}^{\ell_2}(\xbf_i, V/\ \xbf_i) - d_{min}^{\ell_2}(\xbf_i, V/\ \xbf_i)}{d_{min}^{\ell_2}(\xbf_i, V/\ \xbf_i)}  \bigg)=0
\end{align*}


which just means that all data are really far away to each other with the same distance in a very high-dimensional space.

\begin{itemize}
\item As an example, there is a loss of intuition in high-dimension. 
\item You can think that in Normal distribution, in 1-d, most data are concentrated at the center.
\item But in high-diemsnional spaces, most data are concentrated on the surface.
\end{itemize}

\end{columns}

\end{frame}

\item Data have always properties, i.e. structures or invariances, such that they belong to a low-dimensional space called manifold where (geodesic) distances are meaningful.
\end{itemize}

\begin{frame}{Spectral Graph Theory}

\footnotesize

Spectral graph theory (SGT) can

\begin{itemize}
\item find meaningful patterns that reveal multi-scale graph structures.
\item Process data defined on the graph domain (a.k.a. graph signal processing).
\item Boost performance of learning tasks s.a. clustering, classification, recommendation, etc.
\end{itemize}


\vspace{10pt}

\textbf{Graph Laplacian Operator $L$.}
\begin{itemize}
\item Central operator of diffusion processes on graphs.
\item Basis functions of this operator are the well-known Fourier modes
\end{itemize}

 \includegraphics[width=0.55\linewidth]{figures/Laplacian_Fourier.png}


\end{frame}

\begin{frame}{Graph Laplacian}

\begin{itemize}
\item Unnormalized Graph Laplacian:

\begin{align*}
L = D - A
\end{align*}

where $D$ is the degree matrix : $D = diag(d_1, \cdots, d_n)$, and $d_ii = \sum_j A_{ij}$.

\item Normalized Laplacian

\begin{align*}
L_n = D^{-1/2} L D^{-1/2} = I_n - D^{-1/2}A D^{-1/2}
\end{align*}

\item Random Walk Laplacian:

\begin{align*}
L_w = D^{-1}L = I_n - D^{-1}A
\end{align*}

\item All Laplacians are diffusion operators.
\end{itemize}

\end{frame}


\begin{frame}{Graph Spectrum}

\footnotesize

Spectral graph theory can extract the modes of variation of the graph system. This can be done using eigenvalue decomposition.

\begin{tcolorbox}[    enhanced,
    colback=uiCream,
    colframe=embedEmerald,
    sharp corners,
    arc=0pt, 
    boxrule=0.4pt,
    leftrule=3pt,
    % This creates the orange vertical bar on the left:
    borderline west={5pt}{0pt}{embedEmerald}, 
    fontupper=\color{wpDark},
    pad at break*=0mm,
    top=1pt, bottom=2pt, left=5pt, right=5pt]
    
    Considering Laplacian operator as $L$.  The Laplacian is symmetric positive semi-definite, so it admits an eigendecomposition:
    
    \begin{align*}
    L = U \Lambda U^\top  
    \end{align*}
    where $U = [u_1, u_2, \cdots, u_n]$ are eigenvectors (graph Fourier basis) and $\Lambda  = diag(\lambda_1, \lambda_2, \cdots, \lambda_n)$ are eigenvalues with $0 = \lambda_1 \leq \lambda_2 \leq  \cdots \leq \lambda_n$. Also $U^\top U = I_n$. 
    
    \vspace{10pt}
    
    The eigenvectors are the modes of variation.  they capture how signals oscillate across the graph topology, from smooth global trends ($\lambda \approx 0$) to high-frequency local variation ($\lambda$ with a large value).
    
\end{tcolorbox}


\end{frame}

\begin{frame}{Representations for Learning with Graph}


\begin{itemize}
\item Each node $i$ carries a feature vector $\xbf_i$ which gives us $X \in \Rbb^{n \times d}$.
\item Adjacency matrix $A$ tells who is conencted to whom. 
\item Feature matrix $X$ tells what each node knows.
\end{itemize}

So how do we use $X$ and $A$ to learn something?

\vspace{10pt}

Or specifically, how to we prepare their representation that can be fed to a neural network.

\end{frame}

\begin{frame}{Naive Representation}

\footnotesize

\begin{center}
One idea could be to concatenate as [$\xbf_i$, neighbors of $i$]. Collect all neighbor features, concatenate everything, and feed it to a standard neural network.

\vspace{10pt}

\includegraphics[width=0.35\linewidth]{figures/naive_graph_learning.png}
\end{center}

\end{frame}

\begin{frame}{Naive Representation}

\footnotesize

\textbf{But we have problem!}

\vspace{10pt}

A graph has no natural ordering of nodes or neighbors. Permuting the nodes gives the same graph, but a completely different input vector to the MLP.


\vspace{10pt}


\includegraphics[width=0.45\linewidth]{figures/graph_ordering.png}

\textbf{Another issue:} different nodes have different numbers of neighbors. An MLP requires a fixed-size input. It cannot natively handle node $v$ with 2 neighbors and node $u$ with 7 neighbors simultaneously.

\begin{center}

\textbf{We need an operation over neighbor features that is independent of ordering and handles any neighborhood size.} They are called \textbf{A Permutation-Invariant Aggregation}. Example: SUM, MEAN, MAX, etc.
\end{center}


\end{frame}

\begin{frame}{Message Passing Graph Neural Network (Or Spatial GNN)}

\footnotesize

\textbf{Message Passing:}

A mechanism enabling Graph Neural Networks to learn from graph-structured data by having nodes iteratively exchange information with their neighbour's. 

\vspace{5pt}

\begin{itemize}
\item \textbf{Message Creation:} Computing messages (information about features) from neighbouring nodes.
\item \textbf{Message Aggregation:} Aggregating these messages through operations like summation or attention.
\item \textbf{Node Update:} Updating each node’s representation by combining the aggregated information with its current features.
\end{itemize}

\begin{align*}
\hbf_u^{(k+1)} & = UPDATE^{(k)}\bigg( \hbf_u^{(k)}, AGGREGATE^{(k)}( \{\hbf_v^{(k)}, \forall v \in\Ncal(u) \}) \bigg)\\
& =  UPDATE^{(k)}\bigg( \hbf_u^{(k)} , \mbf^{(k)}_{\Ncal(u)} \bigg)
\end{align*}
where $\hbf_u^{(k)}$ is the embedding corresponding to the node $u$, $\Ncal(u)$ is the neighbors of $u$, $\mbf$ is the meassage that is being aggregated.

We use superscripts to distinguish the embeddings and functions at different iterations of message passing. In the context of GNN, they are called different \textbf{layers} of GNN.

\end{frame}

\begin{frame}{Message Passing Graph Neural Network (Or Spatial GNN)}


\footnotesize

\begin{columns}


\column{0.5\linewidth}


Initial embeddings are set as $\hbf_u^{(0)} = \xbf_u \forall u \in Vcal$. After running $k$ iteration,  we can use the output of the final layer to define the embeddings of each node:

\begin{align*}
\zbf_u = \hbf_u^{k}, \quad\forall u \in \Vcal
\end{align*}

\column{0.5\linewidth}

\includegraphics[width=0.65\linewidth]{figures/aggregare_update.png}

\end{columns}


\textbf{Note:} Since the AGGREGATE function takes a set as input, GNNs defined in
this way are permutation equivariant by design.


\end{frame}

\begin{frame}{The Most Basic GNN}


\scriptsize

\begin{tcolorbox}[title={Basic GNN Message Passing}]
\begin{align*}
  \mathbf{h}_u^{(k)} = \sigma\!\bigg(
    \mathbf{W}_{\text{self}}^{(k)}\mathbf{h}_u^{(k-1)}
    + \mathbf{W}_{\text{neigh}}^{(k)}\!\!\sum_{v\in \Ncal(u)}\!\mathbf{h}_v^{(k-1)}
    + \mathbf{b}^{(k)}
  \bigg)
\end{align*}
\end{tcolorbox}

Many GNNs can also be succinctly defined using graph-level equations. In the case of a basic GNN, we can write the graph-level definition of the model as follows:

\vspace{4pt}
\begin{tcolorbox}[title={Graph-level Form}]
\begin{align*}
  H^{(t)} = \sigma\!\Bigl(AH^{(t-1)}\mathbf{W}_{\text{neigh}}^{(t)} + H^{(t-1)}\mathbf{W}_{\text{self}}^{(t)}\Bigr)
\end{align*}
Efficient via \textbf{sparse matrix operations}
\end{tcolorbox}

$\Hbf^{(t)} \in \Rbb^{|V| \times d}$ denotes the matrix of node representations at layer $t$ in the GNN. $\Abf$ is the graph adjacency matrix. This one omits bias for simplicity.

\end{frame}


\begin{frame}{The Most Basic GNN: An Abstract Form}

$$
\mathbf{x}_v^{(k + 1)} = f^{(k + 1)}_{\theta} \left( \mathbf{x}_v^{(k)}, \left\{ \mathbf{x}_w^{(k)} : w \in \mathcal{N}(v) \right\} \right)
$$

where node features $\mathbf{x}_v^{(k)}$ of all nodes $v \in \mathcal{V}$ in a graph $\mathcal{G} = (\mathcal{V}, \mathcal{E})$ are iteratively updated by aggregating localized information from their neighbors $\mathcal{N}(v)$.


\end{frame}

\begin{frame}[fragile]{Neighborhood Normalization}

\footnotesize

\begin{columns}[T]
\column{0.50\textwidth}

\textbf{Problem:} High-degree nodes produce large $\|\sum_v \mathbf{h}_v\|$, causing \emph{instability}.

\vspace{6pt}
\begin{tcolorbox}[defbox,title={Mean Normalization}]
\begin{align*}
  \mathbf{m}_{\Ncal(u)} = \frac{\displaystyle\sum_{v\in\Ncal(u)}\mathbf{h}_v}{|\Ncal(u)|}
\end{align*}
\end{tcolorbox}

\begin{tcolorbox}[keybox,title={Kipf Normalization (GCN)}]
\begin{align*}
  \mathbf{m}_{\Ncal(u)} = \sum_{v\in\Ncal(u)} \frac{\mathbf{h}_v}{\sqrt{|\Ncal(u)||\Ncal(v)|}}
\end{align*}
Down-weights high-degree neighbors
\end{tcolorbox}

\begin{frame}[fragile]{Graph Convolutional Network (GCN)}

\scriptsize

\begin{columns}[T]
\column{0.54\textwidth}

\begin{tcolorbox}[defbox,title={GCN Layer (Kipf \& Welling, 2016)}]
\begin{align*}
  \mathbf{h}_v^{(k)} = \sigma\!\Biggl(
    \mathbf{W}^{(k)}\!\!\!\!\sum_{v\in\N(u)\cup\{u\}}\!\!\!\!
    \frac{\mathbf{h}_v}{\sqrt{|\N(u)||\N(v)|}}
  \Biggr)
\end{align*}
\end{tcolorbox}

\vspace{4pt}
Combines: \textbf{Kipf normalization} + \textbf{self-loop} trick
\begin{align*}
  H^{(t)} = \sigma\!\Bigl((A+I)H^{(t-1)}W^{(t)}\Bigr)
\end{align*}

\begin{tcolorbox}[greenbox,title={Properties}]
\ding{51}~Permutation equivariant\\
\ding{51}~Efficient sparse ops\\
\ding{51}~First-order spectral approximation
\end{tcolorbox}

\begin{frame}[fragile]{Set Aggregators: Going Beyond Sum/Mean}

\scriptsize

\begin{columns}[T]
\column{0.48\textwidth}

\begin{tcolorbox}[defbox,title={Set Pooling (Zaheer et al., 2017)}]
 Universal approximator for permutation-invariant functions:
\begin{align*}
  \mathbf{m}_{\N(u)} = \text{MLP}_\theta\!\!\!\left(\sum_{v\in\N(u)}\!\!\text{MLP}(\mathbf{h}_v)\right)
\end{align*}
\end{tcolorbox}

\begin{tcolorbox}[keybox,title={Janossy Pooling (Murphy et al., 2018)}]
 Apply a \emph{sequence model} on permutations, then average:
\begin{align*}
  \mathbf{m}_{\N(u)} = \text{MLP}_\theta\!\!\left(\frac{1}{|\Pi|}\sum_{\pi\in\Pi}\rho\!\left(\{\mathbf{h}_v\}_\pi\right)\right)
\end{align*}
where $\rho$ is typically an \textbf{LSTM}.
\end{tcolorbox}

\begin{frame}[fragile]{Graph Attention Networks (GAT)}
\scriptsize
\begin{columns}[T]
\column{0.55\textwidth}

\begin{tcolorbox}[defbox,title={Attention-based Aggregation}]

\begin{align*}
  \mathbf{m}_{\N(u)} = \sum_{v\in\N(u)} \alpha_{u,v}\,\mathbf{h}_v
\end{align*}
where the attention weight is:
\begin{align*}
  \alpha_{u,v} = \frac{\exp\!\bigl(\mathbf{a}^\top[\mathbf{W}\mathbf{h}_u \| \mathbf{W}\mathbf{h}_v]\bigr)}{\displaystyle\sum_{v'\in\N(u)}\exp\!\bigl(\mathbf{a}^\top[\mathbf{W}\mathbf{h}_u \| \mathbf{W}\mathbf{h}_v']\bigr)}
\end{align*}
\end{tcolorbox}

\begin{tcolorbox}[keybox,title={Multi-head Attention}]

\vspace{-10pt}

\begin{align*}
  \mathbf{m}_{\N(u)} &= [\mathbf{a}_1 \| \mathbf{a}_2 \| \cdots \| \mathbf{a}_K] \\
  \mathbf{a}_k &= W_k\!\!\sum_{v\in\N(u)}\!\!\alpha_{u,v,k}\,\mathbf{h}_v
\end{align*}
\end{tcolorbox}

\begin{frame}[fragile]{Over-Smoothing: A Core Challenge}

\scriptsize

\begin{columns}[T]
\column{0.50\textwidth}

\begin{tcolorbox}[keybox,title={Over-Smoothing}]
 After many layers, \textbf{all node embeddings converge} to nearly the same vector -- local neighborhood information is lost.
\end{tcolorbox}

\vspace{4pt}
\textbf{Influence Score} (Xu et al., 2018):
\begin{align*}
  I_K(u,v) = \mathbf{1}^\top\!\left(\frac{\partial\,\mathbf{h}_v^{(K)}}{\partial\,\mathbf{h}_u^{(0)}}\right)\!\mathbf{1}
\end{align*}

\begin{tcolorbox}[thmbox,title={Theorem (Xu et al.)}]
 For GCN-style models with normalized aggregation:
\begin{align*}
  I_K(u,v) \;\propto\; p_{G,K}(v \mid u)
\end{align*}
the $K$-step random walk probability from $u$ to $v$.
\end{tcolorbox}

%% --- SKIP CONNECTIONS ------------------------------------------
\begin{frame}[fragile]{Addressing Over-Smoothing: Skip Connections}

\scriptsize

\begin{columns}[T]
\column{0.50\textwidth}

\begin{tcolorbox}[defbox,title={Concatenation Skip (GraphSAGE)}]

\begin{align*}
  \text{UPDATE}_{\text{concat}}(\mathbf{h}_u, \mathbf{m}) =
  \bigl[\text{UPDATE}_{\text{base}}(\mathbf{h}_u, \mathbf{m}) \;\|\; \mathbf{h}_u\bigr]
\end{align*}
Explicitly preserves previous-layer info.
\end{tcolorbox}

\begin{tcolorbox}[keybox,title={Gated Interpolation (Pham et al., 2017)}]

\begin{align*}
  \text{UPDATE}_{\text{interp}} &= \bm{\alpha}_1 \odot \text{UPDATE}_{\text{base}} + \bm{\alpha}_2 \odot \mathbf{h}_u
\end{align*}
$\bm{\alpha}_1 + \bm{\alpha}_2 = \mathbf{1}$; gates learned from data.
\end{tcolorbox}



\column{0.46\textwidth}

\begin{tcolorbox}[thmbox,title={Gated Updates (GRU/LSTM)}]

\begin{align*}
  \mathbf{h}_u^{(k)} = \text{GRU}\!\bigl(\mathbf{h}_u^{(k-1)},\;\mathbf{m}_{\N(u)}^{(k)}\bigr)
\end{align*}
Parameters \textbf{shared across nodes and layers}.
\end{tcolorbox}

\begin{frame}[fragile]{Jumping Knowledge Connections}

\scriptsize

\begin{columns}[T]
\column{0.50\textwidth}

\begin{tcolorbox}[defbox,title={JK Networks (Xu et al., 2018)}]
 Instead of only using the final layer, \textbf{combine all layers}:
\begin{align*}
  \mathbf{z}_u = f_{\text{JK}}\!\bigl(\mathbf{h}_u^{(0)} \| \mathbf{h}_u^{(1)} \| \cdots \| \mathbf{h}_u^{(K)}\bigr)
\end{align*}
\end{tcolorbox}

$f_{\text{JK}}$ can be:
\begin{itemize}
  \item \textbf{Concatenation} (identity)
  \item Max-pooling over layers
  \item LSTM attention over layers
\end{itemize}

\begin{tcolorbox}[greenbox,title={Benefit}]
Each node can \textbf{adaptively select} the right neighborhood size -- some nodes need local info, others global.
\end{tcolorbox}

\begin{frame}[fragile]{Relational Graph Convolutional Networks (R-GCN)}

\scriptsize

\begin{columns}[T]
\column{0.52\textwidth}

\textbf{Challenge:} Graphs with \emph{multiple edge types} (e.g., knowledge graphs).

\begin{tcolorbox}[defbox,title={RGCN Aggregation (Schlichtkrull et al., 2017)}]

\begin{align*}
  \mathbf{m}_{\N(u)} = \sum_{\tau\in\mathcal{R}}
  \sum_{v\in\N_\tau(u)} \frac{W_\tau\,\mathbf{h}_v}{f_n(\N(u),\N(v))}
\end{align*}
One trainable matrix $W_\tau$ \textbf{per relation type}.
\end{tcolorbox}

\begin{tcolorbox}[keybox,title={Parameter Sharing via Basis Decomposition}]

\begin{align*}
  W_\tau = \sum_{b=1}^{B} \alpha_{b,\tau}\,\mathbf{B}_b
\end{align*}
Shared basis matrices $\mathbf{B}_1,\ldots,\mathbf{B}_B$ across all relations.
\end{tcolorbox}

\begin{frame}[fragile]{Graph-Level Embeddings: Pooling Strategies}

\scriptsize


\begin{columns}[T]
\column{0.48\textwidth}

\textbf{Goal:} $\{z_1,\ldots,z_{|V|}\} \longrightarrow z_G$

\vspace{4pt}
\begin{tcolorbox}[defbox,title={Set Pooling}]
 Simple sum / mean:
\begin{align*}
  z_G = \frac{\sum_{v\in V} z_v}{f_n(|V|)}
\end{align*}
Effective for \textbf{small graphs} (e.g., molecules).
\end{tcolorbox}

\begin{tcolorbox}[keybox,title={Attention Pooling (Vinyals et al., 2015)}]
 Iterative attention over nodes:
\begin{align*}
  q_t &= \text{LSTM}(o_{t-1}, q_{t-1}), \quad  a_{v,t} &\propto \exp(f_a(z_v, q_t))\\
  o_t &= \textstyle\sum_v a_{v,t}\,z_v
\end{align*}
$z_G = o_1 \| o_2 \| \cdots \| o_T$
\end{tcolorbox}

\column{0.48\textwidth}

\begin{tcolorbox}[thmbox,title={Graph Coarsening (DiffPool)}]
 Hierarchically collapse nodes into clusters:
\begin{align*}
  S &= f_c(G, Z) \;\in\;\mathbb{R}^{|V|\times c}\\
  A_{\text{new}} &= S^\top A S\\
  X_{\text{new}} &= S^\top Z
\end{align*}
\end{tcolorbox}
