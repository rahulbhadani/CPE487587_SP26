\begin{frame}[fragile]{What is a CNN?}
    \begin{tcolorbox}[colback=mainBlue, colframe=neonBlue, fonttitle=\bfseries, title=Beyond Flattening]
        \color{white}\small
        A \textbf{Convolutional Neural Network (CNN)} is designed for grid-structured data. Unlike FNNs, it treats images as spatial volumes, preserving local relationships.
    \end{tcolorbox}

    \vspace{0.3cm}
    
    \textbf{The Architectural Edge:}
    \begin{itemize}
        \item \textcolor{mainBlue}{\textbf{Local Receptive Fields:}} Neurons only ``see" a small local region, mimicking the human visual cortex.
        \item \textcolor{mainBlue}{\textbf{Spatial Hierarchy:}} Layers learn to recognize edges, then shapes, then objects.
    \end{itemize}

    \vfill
    \centering

\begin{frame}[fragile]{Weight Sharing \& Pooling}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{tcolorbox}[colback=white, colframe=neonBlue, title=\small Weight Sharing, left=1mm]
                \scriptsize
                Filters share weights across the entire image.
                \begin{itemize}
                    \item \textbf{Efficiency:} Drastically reduces parameters.
                    \item \textbf{Feature Detection:} A vertical edge detector works everywhere!
                \end{itemize}
            \end{tcolorbox}
        \end{column}
        
        \begin{column}{0.48\textwidth}
            \begin{tcolorbox}[colback=white, colframe=neonGreen!70!black, title=\small Pooling, left=1mm]
                \scriptsize
                Subsampling to reduce spatial dimensions.
                \begin{itemize}
                    \item \textbf{Invariance:} Stable against small translations/shifts.
                    \item \textbf{Speed:} Lowers computational load \& memory.
                \end{itemize}
            \end{tcolorbox}
        \end{column}
    \end{columns}

        \centering \small
        \textbf{FNN:} 224x224 image $\approx$ 150k+ weights per neuron. \\
        \textbf{CNN:} 3x3 filter = \textcolor{blue}{\textbf{9 weights}} per feature map.
    \end{exampleblock}
\end{frame}


\begin{frame}{Building blocks: Convolution}

  \begin{columns}
        \column{0.4\linewidth}
    \begin{tblock}{}
        Convolution is an operation to reduce the number of pixels through some operation that, in turn, reduces the number of weights to learn.
    \end{tblock}
    In convolution, the reduction in the number of features happens by sliding a kernel across an image, performing the dot product.


\column{0.6\linewidth}
\textbf{Kernel}: It is an $n \times n$ matrix of some numbers whose dimension is usually smaller than the image size.

\begin{frame}[fragile]{A note on multi-dimensionality and vectorization}
    \begin{itemize}
        \item \textbf{High-Dimensionality:} Images are high-dimensional datasets. This is true even more for \textcolor{indigo}{\textbf{colored images}}.
        \item Let $X \in \mathbb{R}^{H \times W}$ be a matrix (for a grayscale image of height $H$ and width $W$).
        \item \textbf{Flattening:} After flattening, an image is represented as a column vector $X \in \mathbb{R}^{D}$:
        \begin{itemize}
            \item \textbf{Grayscale:} $D = H \times W$
            \item \textbf{Color (3 channels):} $D = H \times W \times 3$
        \end{itemize}
    \end{itemize}

\begin{frame}[fragile]{Tensor}
    We need a concept of higher-order matrices called a \textcolor{indigo}{\textbf{Tensor}}.

    \begin{itemize}
        \item $X \in \mathbb{R}^{H \times W \times D}$ is an \textbf{order-3 tensor}. Consider it as $D$ channels of matrices. (Note: $D=1$ reduces it to a standard matrix).
    \end{itemize}

    \begin{tcolorbox}[colback=indigo!5, colframe=indigo, title=Tensor Orders]
        \small
        \begin{itemize}
            \item \textbf{Order-0:} A Scalar value.
            \item \textbf{Order-1:} A Vector.
            \item \textbf{Order-2:} A Matrix. \textit{(Corrected from your text)}
            \item \textbf{Order-3:} A Cube (e.g., RGB Image).
        \end{itemize}
    \end{tcolorbox}
    \small We can also have \textbf{4 dimensions} if we consider opacity (e.g., RGBA in PNG).
\end{frame}

\begin{frame}[fragile]{Tensor to Vector}
    Given a tensor, we can arrange all numbers into a long vector following a \textbf{prespecified order}.

    \vspace{0.3cm}
    \textbf{Example (Matrix Vectorization - Column First Order):}
    \begin{center}
        \renewcommand{\arraystretch}{1.2}
        $A = \begin{bmatrix} \textcolor{indigo}{1} & \textcolor{cyanAccent}{2} \\ \textcolor{indigo}{3} & \textcolor{cyanAccent}{4} \end{bmatrix} \xrightarrow{\text{MATLAB } A(:)} \begin{bmatrix} \textcolor{indigo}{1} \\ \textcolor{indigo}{3} \\ \textcolor{cyanAccent}{2} \\ \textcolor{cyanAccent}{4} \end{bmatrix}$
    \end{center}
    
    \begin{exampleblock}{Recursive Vectorization}
        \small
        To vectorize an \textbf{order-3 tensor}, we vectorize the first channel (matrix), then the second, and so on. This recursive process applies to tensors of any order $> 3$.
    \end{exampleblock}
\end{frame}

\begin{tcolorbox}[colback=white, colframe=blockBlue, arc=1mm, boxrule=0.5pt]
    Basic building blocks of CNN are sequence of \textbf{(i)} Convolutional layer with padding \textbf{(ii)} a nonlinearity such as a ReLU \textbf{(iii)} a pooling layer such as max-pooling to reduce the resolution.
\end{tcolorbox}

\textcolor{red!80!black}{But pooling layer reduces dimension by $\log_2 d$ for $d$ dimension input, and it can happen fast exponentially.}

\begin{enumerate}
    \item \textbf{Simonyan and Zisserman (2014)} used multiple convolutional layer in between downsampling via max-pooling in the form of block.
    \item \textbf{Two $3 \times 3$ convolution} have same effect as one $5\times 5$ convolution layer: \textcolor{logicGreen!70!black}{\textbf{deep and narrow networks significantly outperform their shallow counterparts.}}
\end{enumerate}

\begin{frame}{Network in Network (NiN)}

Two main problems with AlexNet and VGGNet:
\begin{enumerate}
\item FCN at the end of the network has tremendous number of parameters.
\item Cannot add FCN earlier in the network otherwise it can destroy spatial information.
\end{enumerate}
NiN:

\begin{enumerate}
\item use  $1\times 1$ convolutions to add local nonlinearities across the channel activations
\item use global average pooling to integrate across all locations in the last representation layer
\end{enumerate}

\end{frame}

\begin{frame}{Applying Batch Normalization in CNN}

\begin{itemize}
    \item \textcolor{blockBlue}{\textbf{Placement:}} With convolutional layers, we can apply batch normalization after the convolution but before the nonlinear activation function.
    
    \item \textcolor{logicGreen!70!black}{\textbf{Per-Channel Logic:}} The key difference from batch normalization in fully connected layers is that we apply the operation on a per-channel basis across all locations.
    
    \item \textcolor{vggDeepBlue}{\textbf{Dimensions:}} 
    \begin{itemize}
        \item Assume minibatches contain $m$ training samples.
        \item For each channel, output height is $p$ and width is $q$.
        \item Normalization is carried out over the \textbf{$m \cdot p \cdot q$} elements per output channel simultaneously.
    \end{itemize}

    \item \textcolor{red!80!black}{\textbf{Spatial Normalization:}} 
    \begin{itemize}
        \item Collect values over all spatial locations to compute mean and variance.
        \item Apply the same mean/variance within a given channel to every spatial location.
    \end{itemize}

    \item \textcolor{citeGray}{\textbf{Parameters:}} Each channel has its own scale and shift parameters, both of which are scalars.
\end{itemize}

\end{frame}

\begin{frame}{Language Model}

    \begin{block}{Definition}
        A \textbf{language model} is a machine learning model designed to predict upcoming words in a sequence.

        \item It can also assign a joint probability to an \textbf{entire sentence}: $P(w_1, \dots, w_n)$.

\[
p(w_t | w_{t-n+1}, \dots, w_{t-1}) = \frac{\text{count}(w_{t-n+1}, \dots, w_t)}{\text{count}(w_{t-n+1}, \dots, w_{t-1})}
\]

\begin{frame}{How to estimate probabilities}

\textbf{Maximum Likelihood Estimation method:}

Count words from a corpus, and normalize the counts so they fall between 0 and 1.

Computing 2-gram probability:

\begin{align*}
p(w_n|w_{n-1}) = \frac{C(w_{n-1}w_n)}{\sum_w C(w_{n-1} w)} = \frac{C(w_{n-1}w_n)}{ C(w_{n-1})} 
\end{align*}
because the sum of all 2-gram counts that start with a given word $w_{n-1}$ is equal to 1-gram count for that word $w_{n-1}$.


\end{frame}

\begin{columns}[T]

    \begin{column}{0.5\textwidth}
        \textcolor{blockBlue}{\textbf{Relative Frequency Estimation:}}
        
        \vspace{10pt}
        \begin{itemize}
            \item \textbf{Definition:} $n$-gram probability is obtained by dividing the observed frequency of a particular sequence by the observed frequency. 
            
            \vspace{10pt}
            \item \textbf{Terminology:} This approach is called \textcolor{logicGreen!70!black}{\textbf{relative frequency}}.
        \end{itemize}
    \end{column}

    % --- Right Column: General Equation ---

        \textbf{General Equation:}

        \begin{tcolorbox}[colback=vggDeepBlue!5, colframe=vggDeepBlue, arc=1mm, boxrule=0.8pt]
            \small
            \begin{align*}
            P(w_n | w_{n-N + 1: n-1}) = \frac{C(w_{n-N+1:n-1} w_n)}{C(w_{n-N+1:n-1})}
            \end{align*}
        \end{tcolorbox}
        
        \vspace{5pt}
        \scriptsize \textcolor{citeGray}{Estimating the probability of a word given its $N-1$ previous words.}

\begin{frame}{Evaluating LMs}
\vspace{-5pt}
    \begin{columns}[T]
        % --- Left Column: Evaluation Types ---

                Embed into an application and evaluate how much the application improves.
            \end{tcolorbox}
            
            \vspace{2pt}
            \textit{\scriptsize It is comparative: compare two candidate LMs on tasks such as speed recognizer and machine translation.}

            \vspace{8pt}
            
            \begin{tcolorbox}[colback=evalAzure!5, colframe=evalAzure, arc=1mm, title=\scriptsize Intrinsic Evaluation, fonttitle=\bfseries]
                Measuring quality of model independent of application. 
                \vspace{2pt}
                \newline \textbf{Example:} \textcolor{evalAzure}{perplexity}.
            \end{tcolorbox}
        \end{column}

        % --- Right Column: Perplexity Math ---
        \begin{column}{0.48\textwidth}
            \scriptsize
            \textcolor{evalSlate}{\textbf{Perplexity of N-gram Model:}}
            The perplexity of an LM on a test set is the inverse probability of the test set normalized by the number of words or tokens. 
            
            \vspace{4pt}
            We normalize by taking the N-th root. If we have test set $W = w_1 w_2, \cdots w_N$:

            \begin{tcolorbox}[colback=white, colframe=evalSlate!30, arc=1mm]
                \begin{align*}
                perplexity(W) & = \sqrt[N]{\frac{1}{P(w_1 w_2 \cdots w_N)}}\\
                & = \sqrt[N]{\prod_{i=1}^N \frac{1}{P(w_i | w_1  \cdots w_{i-1})}}
                \end{align*}
            \end{tcolorbox}
            \centering \tiny \textit{Using chain rule of probabilities}
        \end{column}
    \end{columns}


    \begin{beamercolorbox}[wd=\textwidth, sep=5pt, rounded=true]{block body}
        \footnotesize \textcolor{evalGold!80!black}{\faExclamationTriangle} \textbf{Caveat:} An intrinsic improvement in perplexity doesn't guarantee an extrinsic improvement in performance.
    \end{beamercolorbox}
\end{frame}

\begin{frame}[fragile]{Perplexity: Weighted Average Branching Factor}


    \begin{columns}[T]
        % --- Left Column: Definitions ---
        \begin{column}{0.47\textwidth}
            \footnotesize
            \begin{beamercolorbox}[wd=\textwidth, sep=4pt, rounded=true]{block body}
                \footnotesize \textcolor{branchEmerald}{\faCheckCircle} Lower perplexity means the model is more confident and accurate.
            \end{beamercolorbox}

            \vspace{5pt}
            \textcolor{branchEmerald}{\textbf{Branching factor of a language:}} 
            The number of possible words that can follow any word. 
            
            \begin{itemize}
                \item \textbf{Deterministic:} Only 1 word follows; factor is 1.
                \item \textbf{Uniform:} $V$ words equally likely; factor is $V$.
                \item \textbf{General:} Average number of possible next words, weighted by their probabilities.
            \end{itemize}

            \vspace{5pt}
            \begin{tcolorbox}[colback=branchEmerald!5, colframe=branchEmerald, arc=1mm, title=\scriptsize Uniform Case, fonttitle=\bfseries]
                If $P(w_i) = \frac{1}{V}$, then:
                \begin{equation*}
                Perplexity = (V^N)^{\frac{1}{N}} = V
                \end{equation*}
            \end{tcolorbox}
        \end{column}

        % --- Right Column: Worked Example ---
        \begin{column}{0.51\textwidth}
            \scriptsize
            \textcolor{branchGold}{\textbf{Example: Bigram Model}} \\
            Test seq: $\text{I} \rightarrow \text{love} \rightarrow \text{NLP}$ ($N=3$) \\
            $P(\text{I}|\langle s \rangle)=1.0, P(\text{love}|\text{I})=0.5, P(\text{NLP}|\text{love})=0.8$

            \vspace{8pt}
            \textbf{\textcolor{branchGold}{Step 1:}} \textbf{Local Branching Factors} \\
            $B_1 = 1/1.0 = 1$ \\
            $B_2 = 1/0.5 = 2$ \\
            $B_3 = 1/0.8 = 1.25$

            \vspace{8pt}
            \textbf{\textcolor{branchGold}{Step 2:}} \textbf{Perplexity (Weighted Avg)} \\
            \begin{tcolorbox}[colback=white, colframe=branchGold!40, arc=1mm]
                \centering
                $\text{PPL} = \sqrt[3]{B_1 \times B_2 \times B_3}$ \\
                $= \sqrt[3]{1 \times 2 \times 1.25} = \sqrt[3]{2.5} \approx \mathbf{1.357}$
            \end{tcolorbox}

            \vspace{5pt}
            \textbf{Interpretation:} \\
            \textit{Avg meaningful next-word choices = 1.357 (not naive vocab size)}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{N-gram Language Models Don't Work Well}
    \vspace{5pt}
    
    \begin{itemize}
        \item \textbf{Data Sparsity:} 
        As $n$ increases, the number of possible $n$-grams grows exponentially, but they might never appear in a training corpus.
        
        \vspace{8pt}
        
        \item  \textbf{The Zero-Probability Problem:} 
        Unseen n-grams receive zero probability, which is problematic. 
        
        \vspace{8pt}
        
        \item  \textbf{Context Limitations:} 
        Long-range dependencies are impossible, or computationally intractable to capture.
        
        \vspace{8pt}
        
        \item 
         \textbf{Lack of Semantic Generalization:} 
        N-grams treat words as discrete symbols and fail to generalize.
    \end{itemize}

  \begin{frame}{Vanishing Gradient / Exploding Gradient Problem}
  As we see that when we unroll the network, we have multiplication of $W_2$ (recurrent weight) with itself many times.
  
  \begin{tblock}{Exploding Gradient}
     If $W_2 > 1$, then eventually, the gradient will become really large ($W_2^{50} \gg 1$).
  \end{tblock}

 \begin{tblock}{Vanishing Gradient}
     If $W_2 < 1$, then eventually, the gradient will become really small ($W_2^{50} \ll 1$).
  \end{tblock}



\end{frame}


\begin{frame}{Vanishing Gradient / Exploding Gradient Problem}
Consider an RNN with ReLU activation:
  \begin{multiequation}
    h_t &= \max(0, W_1 x_t + W_2 h_{t-1} + b_1) \\
    y_t &= W_3 h_t + b_2
  \end{multiequation}
  The backpropagation through time (BPTT) is used for training sequence data.

  The gradient of the loss ($L$) with respect to $W_2$ is:
  $$
  \frac{\partial L}{\partial W_2} = \sum_{k=1}^{t} \frac{\partial L}{\partial y_k} \cdot \frac{\partial y_k}{\partial h_k} \cdot \frac{\partial h_k}{\partial W_2}
  $$
  where $k$ is the index for time.
  
\end{frame}

\begin{frame}{Vanishing Gradient / Exploding Gradient Problem}

  $$
  \frac{\partial h_k}{\partial W_2} = \frac{\partial h_k}{\partial h_{k-1}}\cdot \frac{\partial h_{k-1}}{\partial W_2} = W_2\cdot  \frac{\partial h_{k-1}}{\partial W_2}  = W_2 \cdot \left( W_2 \cdot \frac{\partial h_{k-2}}{\partial W_2} \right) = W_2^2 \cdot \frac{\partial h_{k-2}}{\partial W_2}
  $$
  Continuing this pattern, we get:
  $$
  \frac{\partial h_k}{\partial W_2} = W_2^{k-1} \cdot \frac{\partial h_1}{\partial W_2}
  $$
  This shows that the gradient $\frac{\partial h_k}{\partial W_2}$ is proportional to $W_2$ and the previous gradients.

  \bigskip
  As you see, if $|W_2| > 1$, then the gradient grows exponentially (exploding gradient).

  \bigskip
  If $|W_2| < 1$, then the gradient vanishes.
\end{frame}


\begin{frame}[fragile]{Long-term Dependency Problem in RNN}
    \small
    \begin{beamercolorbox}[wd=\textwidth, sep=6pt, rounded=true]{block body}
         RNN also suffers from \textbf{long-term dependency problem}. When the gap between the relevant information and the point where it is need becomes very large, prediction becomes poorer.
    \end{beamercolorbox}

    \vspace{10pt}

\begin{frame}{What's special in LSTM}
\begin{tblock}{Central Idea}
LSTM Introduces cell state and gating mechanisms to preserve long-term information. A memory cell (interchangeably block) which can maintain its state over time, consisting of an explicit memory (aka the cell state vector) and gating units which regulate the information flow into and out of the memory.
\end{tblock}
 
 There are 3 kinds of gating mechanism:
  \begin{itemize}
    \item \textbf{Forget Gate}: Decides what information to discard
    \item \textbf{Input Gate}: Decides what new information to store
    \item \textbf{Output Gate}: Decides what to output
  \end{itemize}


\end{frame}

\begin{frame}[fragile]{Gating Mechanism in LSTM}

\footnotesize 
    \begin{columns}[T]
        % --- Left Column: Forget Gate Logic ---
        \begin{column}{0.48\textwidth}
            \begin{tcolorbox}[colback=gateKeep!5, colframe=gateKeep, arc=1.5mm, title=\small \textbf{1. Forget Gate} ($f_t$), fonttitle=\bfseries]
                \footnotesize
                Output of $f_t$ tells the percentage the previous cell state will be kept in calculating the updated cell state.
                
                \vspace{2pt}
                \begin{itemize}
                    \item \textbf{0} = forget everything
                    \item \textbf{1} = keep everything
                \end{itemize}
            \end{tcolorbox}
        \end{column}

        % --- Right Column: Input Gate Logic ---
        \begin{column}{0.48\textwidth}
            \begin{tcolorbox}[colback=gateAdd!5, colframe=gateAdd, arc=1.5mm, title=\small \textbf{2. Input Gate} ($i_t$), fonttitle=\bfseries]
                \footnotesize
                Output if $i_t$ tell the percentage of the new candidate that will be used in calculating the updated cell state.
                
                \vspace{2pt}
                \begin{itemize}
                    \item \textbf{0} = ignore new info
                    \item \textbf{1} = accept fully
                \end{itemize}
            \end{tcolorbox}
        \end{column}

    \begin{tcolorbox}[enhanced, title=Byte-pair Encoding, colframe=bpeIndigo, colback=white, attach boxed title to top left={xshift=3mm, yshift=-2mm}, boxed title style={colback=bpeIndigo}]
        \begin{enumerate}
            \item \small \textcolor{bpeIndigo}{\faArrowRight} Start with all individual characters as tokens, (a, b, c, etc.)
            \item \small \textcolor{bpeIndigo}{\faArrowRight} Look through your training data and find the most common pair of adjacent characters
            \item \small \textcolor{bpeIndigo}{\faArrowRight} Merge the most frequent pair of adjacent tokens into a new token and add to your vocabulary
            \item \small \textcolor{bpeIndigo}{\faArrowRight} Repeat steps 2-3 thousands of times until you have your desired vocabulary size
        \end{enumerate}
    \end{tcolorbox}
\end{frame}

\begin{frame}[fragile]{Embedding Layer}

        \item[\textcolor{embedBlue}{\faIcon{vector-square}}] Each word can be represented as a {\color{embedDark}\texttt{dmodel}}-dimensional vector.
        \item[\textcolor{embedGold}{\faIcon{dice}}] Initially, you can use rand function to randomly initialize the embedding matrix.
    \end{itemize}

    \vspace{3pt}

    % Matrix Math Section
    \begin{center}
        \footnotesize
        \begin{equation*}
        \renewcommand{\arraystretch}{1.2}
        % One-Hot Matrix
        \underbrace{
            \begin{bmatrix}
             \cdot & \cdot & \cdot & \cdot \\
             \cdot & \cdot & \cdot & \cdot \\
             \cdot & \cdot & \cdot & \cdot \\
             \cdot & \cdot & \cdot & \cdot \\
             \cdot & \cdot & \cdot & \cdot 
            \end{bmatrix}
        }_{\mathclap{\substack{\text{\textbf{\textcolor{embedBlue}{One Hot Encoding}}} \\ \text{vocabsize} \times \text{vocabsize}}}} 
        \times 
        % Embedding Matrix
        \underbrace{
            \begin{bmatrix}
            \cdot & \cdot & \cdot \\
            \cdot & \cdot & \cdot \\
            \cdot & \cdot & \cdot \\
            \cdot & \cdot & \cdot \\
            \cdot & \cdot & \cdot 
            \end{bmatrix}
        }_{\mathclap{\substack{\text{\textbf{\textcolor{embedGold}{Embedding Matrix}}} \\ \text{vocabsize} \times \text{dmodel}}}} 
     \quad \quad \quad   = 
        % Result Matrix
        \underbrace{
            \begin{bmatrix}
             \cdot & \cdot & \cdot \\
             \cdot & \cdot & \cdot \\
             \cdot & \cdot & \cdot \\
             \cdot & \cdot & \cdot \\
             \cdot & \cdot & \cdot 
            \end{bmatrix}
        }_{\mathclap{\substack{\text{\textbf{Embedding for the sequence}} \\ \text{vocabsize} \times \text{dmodel}}}}
        \end{equation*}
    \end{center}

    \begin{flushright}
        \footnotesize \textcolor{darkgray}{Note: Matrix multiplication acts as a "lookup" for specific row vectors.}
    \end{flushright}

        PE = \begin{bmatrix} \cdot & \cdot & \cdot \\ \cdot & \cdot & \cdot \\ \cdot & \cdot & \cdot \\ \cdot & \cdot & \cdot \\ \cdot & \cdot & \cdot \end{bmatrix}
        \end{equation*}
        
        
    \end{column}

    % Column 2: The Explanation
    \begin{column}{0.8\textwidth}
        \scriptsize
       \begin{itemize}
       
       \item  As transformer itself doesn't provide information about positions, we have to encode them separately.
       
       \item  \textbf{Positional Encoding} is used to provide a relative position for each token or word in a sequence. When reading a sentence, each word is dependent on the word around it. Some words have different meaning in different context.
       
       \item  To encode the position, we use cosine, and sine.
        
        \begin{tikzpicture}
            \begin{axis}[
                width=4cm, height=2.5cm,
                axis lines=none,
                xmin=0, xmax=10, ymin=-1.2, ymax=1.2,
                samples=50, domain=0:10
            ]
                \addplot[blue, thick] {sin(deg(x))};
                \addplot[red, dashed] {cos(deg(x))};
            \end{axis}
        \end{tikzpicture}
       
\end{itemize}       
        
       

       
        

        Consider $L$ as the maximum number of positions or vectors in the embedding.

        

        \begin{tcolorbox}[colback=black!5, colframe=black!50, arc=1mm, boxrule=0.5pt, left=2pt, right=2pt, top=2pt, bottom=2pt]
        \tiny\begin{verbatim}
for each k = 0 to L-1
  for each i = 0 to dmodel/2
    PE(k, 2i) = sin( k /n^{2i/dmodel})
    PE(k, 2i+1) = cos( k /n^{2i/dmodel}) \end{verbatim}
        \end{tcolorbox}
        
        $n$ is user-defined, set as $10000$ by the original author.
        
        Once PE Matrix is calculated, it is added to the embedding matrix calculated in the previous step.
    \end{column}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Layer Normalization}
\begin{columns}[T]
    \begin{column}{0.48\textwidth}
        \begin{tcolorbox}[colback=LNDark, colframe=LNDark, arc=1mm, fontupper=\color{white}]
            \small \faArrowRight \ Input to Layer Norm is a single vector of dimensioanlity $d_{model}$ and the output is that vector normalized, of the same dimensionality.
        \end{tcolorbox}

        \vspace{5pt}
        \scriptsize
        \begin{enumerate}
            \item Calculate the mean $\mu$ and the standard deviation $\sigma$ over the elements of the vector to be normalized.
        \end{enumerate}

        \begin{tcolorbox}[colback=white, colframe=LNEmerald!30, title=\tiny, fonttitle=\bfseries]
            \tiny If $\mathbf{x}$ is the embedding, then
            \begin{align*}
            \mu = \cfrac{1}{d_{model}}\sum_{i=1}^{d_{model}} x_i \\
            \sigma = \sqrt{\cfrac{1}{d_{model}}\sum_{i=1}^{d_{model}} (x_i - \mu)^2 }
            \end{align*}
        \end{tcolorbox}
    \end{column}

    \begin{column}{0.48\textwidth}
        \scriptsize
        \begin{enumerate}
            \setcounter{enumi}{1}
            \item Then the normalized output is 
        \end{enumerate}

        \begin{tcolorbox}[enhanced, colback=white, colframe=LNEmerald!50, borderline west={2pt}{0pt}{LNEmerald}]
            \begin{align*}
            \hat{\mathbf{x}} = \frac{\mathbf{x} - \mu }{\sigma}
            \end{align*}
        \end{tcolorbox}

        \vspace{5pt}
        \begin{tcolorbox}[enhanced, colback=white, colframe=LNEmerald, boxrule=1pt, arc=2mm, title=\centering \tiny ]
            \tiny In the implementation, we have two learnable parameters $\gamma$ and $\beta$ such that
            \begin{align*}
            LayerNorm(\mathbf{x}) =\gamma \frac{\mathbf{x} - \mu }{\sigma} + \beta
            \end{align*}
        \end{tcolorbox}
    \end{column}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Language Model Head}




    \begin{columns}[T]
        % --- Left Column: Text & Theory ---
        % Left Column: Primary Concept
        \begin{column}{0.44\textwidth}
            \begin{tcolorbox}[modernbox]
                The language model head is the last component of the transformer.
            \end{tcolorbox}
            
            \vspace{0.1cm}
            
            \tiny
            
            The job of the language model is to take the output of the final transformer layer from the last token $N$ and use it to predict the upcoming word at position $N+1$.
            
            \vspace{0.1cm}
            
            \begin{description}
            
                \item[Linear Layer] There is a linear layer that takes output $h_N^L$ which is the output token embedding at position$N$ from the final block $L$ of shape $1 \times d_{model}$ to the logit vector (score vector) and then have a single score for each of the $|V|$ possible words in the vocabulary $V$. The logit vector has dimensionality $1\times |V|$.
                \item[Unembedding Layer] Unembedding layer $E^\top$ where $E$ is the embedding matrix of shape $|V| \times d_{model}$, with same weight is used to map back from the embedding shape $1 \times d_{model}$ to $1\times |V|$.

                \item[Softmax] Turns the logits into probabilities.
            \end{description}
