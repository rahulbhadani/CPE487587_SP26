\begin{frame}{Lecture Overview}
    \begin{itemize}
        \item In this lecture we discussed generative models (briefly) and how to model distributions of data using density estimation techniques. One advantage to these generative models is that we can generate data from the distribution once we have the model.   
        \item Our goal in this lecture is to introduce {\em \color{MediumRed}Generative Adversarial Networks} (GANs) for generating data from a distribution $p_{data}(\x)$. GANs are able to model complex distributions of data that would be incredible challenging to model using the density estimation approaches discussed in previous lectures. 
        \item Modeling $p_{data}(\x)$ is increasingly challenging for complex data, such as real scenes, people, etc. 
    \end{itemize}
\end{frame}

\begin{frame}{Discriminative vs. Generative Models}
    \begin{columns}[T]
        \begin{column}{0.49\textwidth}
            \centering
            \begin{block}{Discriminative}
                \small Focuses on the \textbf{boundary} between classes.
                \vspace{0.5em}
                \begin{itemize}
                    \item Maps high-dimensional input $\mathbf{x}$ to label $y$.
                    \item Learns $P(y \mid \mathbf{x})$.
                    \item \textit{Goal:} Classification and Regression.
                \end{itemize}
            \end{block}
        \end{column}
        
        \begin{column}{0.49\textwidth}
            \centering
            \begin{block}{Generative}
                \small Focuses on the \textbf{distribution} of the data.
                \vspace{0.5em}
                \begin{itemize}
                    \item Models the process that \textit{creates} the data.
                    \item Learns $P(\mathbf{x}, y)$ or $P(\mathbf{x})$.
                    \item \textit{Strategy:} MLE or Variational Inference.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}

    \vspace{1.5em}

    \begin{quote}
        \centering
        \small
        While \textbf{Discriminative} models tell you how to tell things apart, \textbf{Generative} models give you the "recipe" for the data itself.
    \end{quote}
    	\centering
    	
    \textbf{	GAN is a generative model.}
    	
\end{frame}

\begin{frame}{Generative Adversarial Net}
    \small
\begin{block}{}
\begin{itemize}
    \item We have two neural network in a GAN
    \begin{itemize}
        \item The network $G$'s goal is to generate sample that look like they came from $p_{data}(\x)$. 
        \item The network $D$'s goal is to effectively discriminate between real samples from $p_{data}(\x)$ and fake samples from $G$. 
    \end{itemize}
    \item The goals of $G$ and $D$ are different! We can view this scenario as a game between $D$ and $G$. 
\end{itemize}
\end{block}

\begin{quote}
We train $D$ to maximize the probability of assigning the
correct label to both training examples and samples from $G$. We simultaneously train $G$ to minimize $\log(1 - D(G(z)))$.
\end{quote}


\end{frame}





%------------------------------------------------
\begin{frame}[fragile]{Formally Defining the Optimization Task}

\begin{tblock}{}
The optimization of $\theta_g$ and $\theta_d$ can be cast as a $\min$-$\max$ task, which is given by
\begin{multiequation}
\min_{G}\max_{D} V (D, G) = \left\{ \Ebb_{\xbf\sim p_{data}}\left[ \log(D(\x)) \right] + \Ebb_{\zbf\sim p_{\zbf}}\left[\log(1 - D(G(\zbf)))\right]\right\}
\end{multiequation}

$V$ is called as value function.

\end{tblock}

From the original paper: alternate between $k$ steps of optimizing $D$ and one step of optimizing $G$.

\begin{texample}{}
 
 In the beginning $G$ is poor, so $D$ will reject samples with a high confidence, and 
 $\log(1 - D(G(z)))$ saturates. So, rather than training $G$ to minimize $\log(1 - D(G(z)))$ , $G$ is trained to maximize $\log D(G(z))$.
 
\end{texample}


\end{frame}





%------------------------------------------------
\begin{frame}{Formally Defining the Optimization Task}
    
\begin{block}{}
\begin{itemize}
    \item If $\x \sim p_{data}(\x)$ then the discriminator should maximize $\log D(\x)$
    \item If $\zbf\sim p_g(\zbf)$ then the generator seeks to generate a sample that makes the discriminator think it is a normal sample. 
    \item We can use stochastic gradient descent and ascent to adjust the parameters of the network. 
\end{itemize}
\end{block}

\end{frame}

\begin{frame}{Global Optimality of $p_g = p_{\textbf{data}}$}

If we fix the generator $G$, the optimal discriminator $D$ is

\LARGE 

\begin{align*}
\centering
D^*_G(\xbf) = \frac{p_{data}(\xbf)}{p_{data}(\xbf) + p_g(\xbf)}
\end{align*}

\end{frame}


\begin{frame}{Minimax Equation with Optimal Discriminator}

For a fixed G, the minimax is

\begin{align*}
C(G) & = \max_{D} V (D, G) \\
& =   \Ebb_{\xbf\sim p_{data}}\left[ \log(D_G^*(\x)) \right] + \Ebb_{\zbf\sim p_{\zbf}}\left[\log(1 - D_G^*(G(\zbf)))\right]\\
& =  \Ebb_{\xbf\sim p_{data}}\left[ \log(D_G^*(\x)) \right] + \Ebb_{\xbf\sim p_g}\left[\log(1 - D_G^*(\xbf))\right]\\
& = \Ebb_{\xbf\sim p_{data}}\left[ \log   \frac{p_{data}(\xbf)}{p_{data}(\xbf) + p_g(\xbf)} \right] + \Ebb_{\xbf\sim p_g}\left[\log  \frac{p_{\xbf}(\xbf)}{p_{data}(\xbf) + p_g(\xbf)} \right]
\end{align*}

Here, training objective of $D$ can be interpreted as maximizing the log-likelihood for estimating the conditional probability $P(Y = y|\xbf)$ where $Y$ is the random variable denoting where $\xbf$ comes from $p_{data}$ with $y=1$ or from $p_g$ with $y=0$.

\end{frame}


%------------------------------------------------
\begin{frame}{Formally Defining the Optimization Task}
\begin{alertblock}{}
Theorem 1. The global minimum of the virtual training criterion $C(G)$ is achieved if and only if $p_g = p_{data}$. At that point, $C(G)$ achieves the value $-\log (4)$. This point is a Nash Equilibrium. 
\end{alertblock}

\end{frame}

\begin{frame}{Pseudo Code for Training a GAN}

    \footnotesize
\begin{itemize}
    \item[ ] {\bf for} $t = 1, \ldots, T$
    \begin{itemize}
        \item[ ]  {\bf for} $k = 1, \ldots, K$
        \begin{itemize}
            \item Sample $m$ noise data $\zbf_1, \ldots, \zbf_m$ from $p_g(\zbf)$
            \item Sample $m$ data samples $\xbf_1, \ldots, \xbf_m$ from $p_{data}(\x)$
            \item  Update the discriminator by ascending its stochastic gradient:
            \begin{multiequation}
                \nabla_{\theta_d}\left\{ \frac{1}{m}\sum_{i=1}^m \left[ \log(D(\x_i)) + \log(1 - D(G(\zbf_i))) \right]\right\}
            \end{multiequation}
        \end{itemize}
        \item[ ] {\bf end for}
        \item  Sample $m$ noise data $\zbf_1, \ldots, \zbf_m$ from $p_g(\zbf)$
        \item Update the generator by descending its stochastic gradient:
        \begin{multiequation}
            \nabla_{\theta_g} \left\{  \frac{1}{m}\sum_{i=1}^m \log(1 - D(G(\zbf_i)))  \right\}
        \end{multiequation}
    \end{itemize}
    \item[ ] {\bf end for}
\end{itemize}

\end{frame}

\begin{frame}{On Training GANs}
\small

\begin{block}{Challenge}
One issue associated with training a GAN is the form of the loss. Observe above that the generator weights only get updated by the gradients of the second term:
\begin{multiequation}
\log(1 - D(G(\zbf)))
\end{multiequation}\vspace{-2em}
\begin{itemize}
    \item The problem with this is that if the generator sample is really bad, then the discriminator’s prediction is close to zero, and since $\log(1-D)$ is very flat when $D\approx0$ there is not enough ``signal'' to move the generator weights meaningfully. 
    \item Increasing learning rates do not seem to help. This is called the {\em saturation problem} in GANs. 
    \item To fix this, while updating generator weights, it is common to heuristically replace the second term in the GAN loss with $-\log D(G(\zbf))$
\end{itemize}
\end{block}


\end{frame}



%------------------------------------------------
\begin{frame}{On Training GANs}

    \small

\begin{tblock}{Challenge}
Stably training GANs was a challenge faced by the community for quite some time (and continues to be a challenge), and a common resolution is to use {\em Wasserstein GANs. }
\begin{itemize}
    \item The GAN loss is a  form of distance between probability distributions (called the Jensen-Shannon divergence), and this can be generated to other distances. 
    \item A common alternative is the Earth-mover or Wasserstein distance, leading to a different type of GAN model called Wasserstein GAN. The WGAN was shown to improve stability of the optimization process
\end{itemize}
\end{tblock}


\end{frame}

\begin{frame}{Training Loss in GAN Implementation}
\scriptsize

\begin{tcolorbox}[colback=blue!5, colframe=theoryblue, arc=2pt]
    Training loss is actually just Binary Cross Entropy loss effectively.
    $$\mathcal{L}_{BCE}(y, \hat{y}) = -[y \cdot \log(\hat{y}) + (1-y) \cdot \log(1-\hat{y})]$$
\end{tcolorbox}

\begin{columns}[t]
    \begin{column}{0.48\textwidth}
        \textbf{Case 1: Real Sample ($y=1$)}
        \begin{tcolorbox}[colback=gray!5, colframe=gray!50, arc=0pt]
            $$\mathcal{L}_{BCE} = -\log D(x)$$
        \end{tcolorbox}
    \end{column}
    \begin{column}{0.48\textwidth}
        \textbf{Case 2: Fake Sample ($y=0$)}
        \begin{tcolorbox}[colback=gray!5, colframe=gray!50, arc=0pt]
            $$\mathcal{L}_{BCE} = -\log(1 - D(G(z)))$$
        \end{tcolorbox}
    \end{column}

\begin{frame}{Issues with GANs (Google's Comments)}
\begin{tblock}{}
\begin{itemize}
    \item Research has suggested that if your discriminator is too good, then generator training can fail due to vanishing gradients. In effect, an optimal discriminator doesn't provide enough information for the generator to make progress.
    \begin{itemize}
        \item The Wasserstein loss is designed to prevent vanishing gradients even when you train the discriminator to optimality. 
    \end{itemize}
    \item The generator may learn to produce a single output. The generator is always trying to find the one output that seems most plausible to the discriminator. This form of GAN failure is called {\em mode collapse}.
    \item GANs frequently fail to converge, as discussed in the module on training.
\end{itemize}
\end{tblock}
\end{frame}

\begin{frame}{Deep Convolution GAN}
\scriptsize

\begin{tcolorbox}[colback=mainblue!5, colframe=mainblue, arc=3pt, boxrule=0.5pt]
    \textbf{Fractionally-Strided Convolutional Layer} (or Transposed Convolution) is the opposite of the convolution operation.
\end{tcolorbox}

\begin{columns}[t]
    \begin{column}{0.5\textwidth}
        \textbf{Key Concepts:}
        \begin{itemize}
            \item \textbf{Standard Convolution:} In a convolution operation (for example, stride = 2), we obtain a downsampled (smaller) output of the larger input.
            \item \textbf{Fractional Stride:} In fractionally-strided convolution, an upsampled (larger) output is obtained from a smaller input.
        \end{itemize}
        
         
        \begin{tcolorbox}[colback=gray!10, boxrule=0pt, frame hidden, left=2pt, right=2pt, top=2pt, bottom=2pt]
    \tiny \textcolor{gray}{Radford, Alec, Luke Metz, and Soumith Chintala. "Unsupervised representation learning with deep convolutional generative adversarial networks." arXiv preprint arXiv:1511.06434 (2015).}
\end{tcolorbox}

        
      

    \end{column}
    
    \begin{column}{0.48\textwidth}
        \textbf{Mathematical Example:}
        Consider $X = \begin{bmatrix}
        x_{00} & x_{01}\\
        x_{10} & x_{11}
        \end{bmatrix}$ and filter as $K = \begin{bmatrix}
        k_{00} & k_{01} & k_{02}\\
        k_{10} & k_{11} & k_{12}\\
        k_{20} & k_{21} & k_{22}\\
        \end{bmatrix}$ 
       
  \begin{tcolorbox}[colback=lightbg, colframe=gray!50,  fonttitle=\bfseries]
    We insert zeros between every other input pixel. Hence:
    \[
    X = \begin{bmatrix}
    x_{00} & 0 & x_{01} & 0\\
    0 & 0 & 0 & 0\\
    x_{10} & 0 & x_{11} & 0\\
    0 & 0 & 0 & 0\\
    \end{bmatrix}
    \]
    Then we apply regular convolution operation.
\end{tcolorbox}

    \end{column}
\end{columns}



\end{frame}

\begin{frame}{Deep Convolution GAN}

From the original paper:

\begin{enumerate}
\item  Replace any pooling layers with strided convolutions (discriminator) and fractional-strided convolutions (generator).
\item Use batchnorm in both the generator and the discriminator.
\item Remove fully connected hidden layers for deeper architectures.
\item Use ReLU activation in generator for all layers except for the output, which uses Tanh.
\item  Use LeakyReLU activation in the discriminator for all layers.
\end{enumerate}

\begin{frame}{Cycle GAN Architecture}

It is used for image to image translation to transfer some interesting features from image to another image, as we saw earlier. Examples: Horse to Zebra, Summer to Winter, etc.

Cycle GAN introduces \textbf{Cycle Consistency Loss} in addition to adversarial Loss seen in the original GAN.

Cycle GAN has four networks:

\begin{enumerate}
\item $G_{A\to B}$ : a generator for translating images from domain $ A \to B$.

\item $D_B$: a discriminator if an image is real or fake in $B$.

\item $G_{B\to A}$ : a generator for translating images from domain $ B \to A$.

\item $D_A$: a discriminator if an image is real or fake in $A$.

  
        \begin{tcolorbox}[colback=gray!10, boxrule=0pt, frame hidden, left=0pt, right=0pt, top=2pt, bottom=2pt]
    \tiny \textcolor{gray}{Zhu, Jun-Yan, Taesung Park, Phillip Isola, and Alexei A. Efros. "Unpaired image-to-image translation using cycle-consistent adversarial networks." In Proceedings of the IEEE international conference on computer vision, pp. 2223-2232. 2017. \url{https://arxiv.org/pdf/1703.10593}}
\end{tcolorbox}



\end{enumerate}



\end{frame}

                \vspace{-1.5em}
                \begin{equation*}
                \Ebb_{y\sim p_{y}}[\log D_Y(y)] + \Ebb_{x\sim p_{x}}[\log (1 - D_Y(G(x)))]
                \end{equation*}

                \hrulefill
                
                Adversarial Loss for $F: Y \to X$:
                \begin{equation*}
                \Lcal_{GAN}(F, D_X, Y, X) = 
                \end{equation*}
                \vspace{-1.5em}
                \begin{equation*}
                \Ebb_{x\sim p_{x}}[\log D_X(x)] + \Ebb_{y\sim p_{y}}[\log (1 - D_X(G(y)))]
                \end{equation*}
            \end{tcolorbox}
        \end{column}

        % --- Right Column: Cycle Consistency ---
        \begin{column}{0.49\textwidth}
            \begin{tcolorbox}[lossbox=Cycle Consistency Loss, colframe=CycleGreen]
                \scriptsize
                As our goal in cycle consistency is $x \to G(x) \to F(G(x)) \approx x$, we have forward cycle consistency loss. Similarly, a backward cycle consistency requires $y \to F(y) \to G(F(y)) \approx y$.
                

                Cycle consistency loss is
                \begin{equation*}
                \Lcal_{cyc}(G, F) = 
                \end{equation*}
                \vspace{-1.5em}
                \begin{equation*}
                \Ebb_{x\sim p_{x}}[ || F(G(x)) - x||_1] + \Ebb_{y\sim p_{y}}[ || G(F(y)) - y||_1]
                \end{equation*}
                It captures differences of pixel values among generated and original images.
            \end{tcolorbox}
        \end{column}
    \end{columns}



    % --- Bottom Section: Full Objective ---
    \begin{tcolorbox}[finalbox]
        \scriptsize
        Full loss function is $
        \Lcal(G, F, D_X, D_Y) = \Lcal_{GAN}(G, D_Y, X, Y) + \Lcal_{GAN}(F, D_X, Y, X) + \lambda \Lcal_{cyc}(G, F) $
        
        and objective is 
        \begin{equation*}
        G^{*}, F^{*} = \arg\min_{G,F} \max_{D_{x},D_{Y}} \mathcal{L}(G, F, D_{X}, D_{Y}).
        \end{equation*}
    \end{tcolorbox}
\end{frame}

\textbf{Generator:}

\begin{enumerate}
\item Three convolutions, several residual blocks
\item Fractionally-strided convolution with stride of $\frac{1}{2}$
\item One convolution to map features to RGB.

\item Use normalization
\end{enumerate}

\begin{enumerate}
\item Use $70\times 70$ PatchGANs to classify whether70 × 70 overlapping image patches are real or fake.
\item The PatchGAN discriminator is a fully convolutional network (FCN) with 4 layers of 4×4 convolutions (stride 2) — no fully connected (FC) layers.
\item Patch-level discriminator architecture has fewer parameters than a full-image discriminator and can work on arbitrarilysized images in a fully convolutional fashion
\end{enumerate}

\end{frame}

\begin{frame}{Distance and Divergences as Measures of Closeness}
    \begin{columns}[T]
        % --- Column 1: Classical Metrics ---
        \begin{column}{0.48\textwidth}
            \begin{tcolorbox}[enhanced, colback=blue!5, colframe=blue!60!black, title=\small Total Variation (TV), fonttitle=\bfseries, boxrule=0.5pt, arc=2pt]
                \scriptsize
                $\delta (P_r, P_g) = \sup_{A\in \Sigma}[ P_r(A) - P_g(A)]$
                \vspace{0.1cm}
                \hrule
                \vspace{0.1cm}
                \textbf{Note:} The \textbf{sup} finds the least upper bound. It's the "maximum" generalized to cases where a max might not exist. E.g., no max in $(0,1)$ but sup is $1$.
            \end{tcolorbox}

            \begin{tcolorbox}[enhanced, colback=red!5, colframe=red!60!black, title=\small Kullback-Leibler (KL), fonttitle=\bfseries, boxrule=0.5pt]
                \scriptsize
                $KL(P_r \| P_g) = \int \log\left(\frac{P_r(x)}{P_g(x)}\right) P_r(x) \, d\mu(x)$
                \vspace{0.1cm}
                \textit{Assumes absolute continuity.} KL is asymmetric and can be $\infty$ if $P_g(x)=0$ while $P_r(x)>0$.
            \end{tcolorbox}

    Further, Wasserstein Distance shown earlier is intractable as the set $\Pi(P_r, P_g)$ is infinitely large and high-dimensioanl data makes $\gamma$, and infimum requires solving Combinatorial Optimization Problem which is very difficult. Hence, instead, we opt for a dual version of Wasserstein distance as per for \textbf{Kantorovic-Rubinstein duality}:
    
    \begin{align*}
      W(P_r, P_\theta) = \sup_{||f||_L \leq 1}\Ebb_{x\sim P_r}[f(x)] - \Ebb_{x\sim P_\theta}[f(x)]
    \end{align*}
  With this dual, Instead of optimizing over an infinite set of transport plans $\gamma$, we optimize over a neural network  $f$ (the Critic) — a finite-dimensional parameterization of 1-Lipschitz functions. The dual problem is differentiable (neural networks are differentiable), so we can use gradient descent to train the Critic.
  \end{tcolorbox}

\begin{frame}{1-Lipschitz Function}

\footnotesize

\begin{definition}[1-Lipschitz]
A function $f: \mathcal{X} \to \mathbb{R}$ is 1-Lipschitz if
\begin{align*}
\forall x_1, x_2 \in \mathcal{X}: \quad
|f(x_2) - f(x_1)| \le \|x_2 - x_1\|.
\end{align*}
\end{definition}

\vspace{0.3cm}
\begin{itemize}
    \item Output changes \textbf{no faster} than input changes.
    \item If differentiable: $\|\nabla f(x)\| \le 1$ everywhere.  % Fixed: added opening \| and math delimiters
    \item Ensures the Critic $f$ is \textbf{stable and well-behaved}.
    \item Required for the \textbf{Kantorovich--Rubinstein duality} in WGAN.
\end{itemize}

\begin{frame}[fragile]{WGAN Algorithm}
    \begin{tcolorbox}[
        enhanced,
        colback=white,
        colframe=WGANBlue,
        title=\small Algorithm,
        fonttitle=\bfseries,
        boxrule=0.8pt,
        arc=2pt
    ]
    \scriptsize
    % Define the custom command to fix the undefined control sequence
    \SetKwInOut{Require}{Require} 
    
    \begin{algorithm}[H]
        \Require{$\alpha$ (LR), $c$ (clipping), $m$ (batch), $n_{\text{critic}}$ (no. of iterations of the critic per generator iteration)}
        \Require{$w_0$ (initial critic), $\theta_0$ (initial generator's parameter)}
        
        \While{$\theta$ not converged}{
            \For{$t = 0, \dots, n_{\text{critic}}$}{
                Sample $\{x^{(i)}\}_{i=1}^m \sim \mathbb{P}_r$ a batch from the real data\;
                Sample $\{z^{(i)}\}_{i=1}^m \sim p(z)$ a batch of prior samples\;
                $g_w \leftarrow \nabla_w [\frac{1}{m}\sum_{i=1}^m f_w(x^{(i)}) - \frac{1}{m}\sum_{i=1}^m f_w(g_\theta(z^{(i)}))]$\;
                $w \leftarrow w + \alpha \cdot \text{RMSProp}(w, g_w)$\;
                $w \leftarrow \text{clip}(w, -c, c)$ \textcolor{WGANOrange}{\footnotesize $\leftarrow$ \textit{Lipschitz}}\;
            }
            Sample $\{z^{(i)}\}_{i=1}^m \sim p(z)$\;
            $g_\theta \leftarrow - \nabla_\theta  \frac{1}{m}\sum_{i=1}^m f_w(g_\theta(z^{(i)}))$\;
            $\theta \leftarrow \theta - \alpha \cdot \text{RMSProp}(\theta,g_\theta)$\;
        }
    \end{algorithm}
    \end{tcolorbox}

    \begin{itemize}
        \item \tiny All experiments used $\alpha = 0.00005, c = 0.01, m = 64, n_{\text{critic}} = 5$.
    \end{itemize}
\end{frame}

\begin{frame}{Privacy issues with GANs}

\scriptsize

GANs are trained to \textbf{model the real data distribution}.

But in doing so, they can:
\begin{itemize}
    \item \textbf{Memorize} training samples (faces, medical data, text)
    \item \textbf{Reconstruct} exact or near‑exact training data
    \item Leak \textbf{sensitive attributes}: gender, disease, location, ID
    \item Comply with no privacy by default
\end{itemize}

\begin{tcolorbox}[]
GANs \textbf{implicitly store information} about training data
in the generator and discriminator weights.
\end{tcolorbox}

\textbf{Examples of leakage:}
\begin{itemize}
    \item Face GANs: reproduce exact faces from training set
    \item Medical GANs: reveal unique patient biomarkers
    \item Text GANs: regurgitate private sentences
\end{itemize}

\textbf{Membership Inference Attacks:}

Given a sample $x$, attacker asks: Was $x$  in the training set?

How it works:
\begin{itemize}
    \item Use discriminator confidence $D(x)$
    \item Training samples have \textbf{unusually high/low confidence}
    \item Attacker trains a classifier to detect this
\end{itemize}

\end{frame}
